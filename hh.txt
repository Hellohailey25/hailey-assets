/* =========================================================================
   Кнопка «В корзину» на карточке товара.
   Было: жмёшь сколько угодно — один и тот же размер копится в корзине,
   а кнопка через 1,4 с возвращалась в исходный вид, будто ничего не произошло.
   Стало: размер уже в корзине → кнопка «Перейти в корзину» и открывает её.
   Родной обработчик блока не трогаем: перехватываем клик в фазе захвата
   и не пускаем его дальше, когда добавлять уже нечего.
   ========================================================================= */
function hpBuyBtnInit(){
  var btn=document.querySelector('.hp #addBtn'), box=document.querySelector('.hp #sizes');
  if(!btn||!box) return;                                  // не карточка товара
  var art=(document.querySelector('.hp .art')||{}).textContent||'';
  var id=(art.match(/\d{6,}/)||[])[0]||'';
  var LBL=btn.textContent;

  function inCart(sz){
    try{
      return (JSON.parse(localStorage.getItem('hailey_cart_v1'))||[]).some(function(i){
        return String(i.id)===String(id) && String(i.size)===String(sz);
      });
    }catch(e){ return false; }
  }
  function sync(){
    var on=box.querySelector('.sz.on');
    var go=!!(on && inCart(on.textContent.trim()));
    btn.classList.toggle('gocart',go);
    btn.textContent=go?'Перейти в корзину →':LBL;
  }
  btn.addEventListener('click',function(e){
    if(!btn.classList.contains('gocart')) return;         // обычный клик — пусть кладёт в корзину
    e.preventDefault(); e.stopImmediatePropagation();
    var c=document.getElementById('hhCart'); if(c)c.click();
  },true);
  // родной обработчик 1,4 с держит «✓ Добавлено» — свою подпись ставим после него
  btn.addEventListener('click',function(){ setTimeout(sync,1500); });
  box.addEventListener('click',function(){ setTimeout(sync,20); });
  window.addEventListener('storage',function(e){ if(e.key==='hailey_cart_v1') sync(); });
  setInterval(sync,1500);                                 // удалили товар в корзине — кнопка вернётся
  sync();
}
if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',hpBuyBtnInit);
else hpBuyBtnInit();
