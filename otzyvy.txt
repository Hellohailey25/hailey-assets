<script>
(function(){
  var SRC='https://raw.githubusercontent.com/Hellohailey25/hailey-assets/main/reviews.json';
  var PAGE=24, shown=0, filtered=[], DATA=null, cur='';
  var $=function(id){return document.getElementById(id);};

  function stars(n,cls){
    var h='<div class="'+(cls||'stars')+'">';
    for(var i=0;i<5;i++)h+='<svg viewBox="0 0 24 24"'+(i<n?'':' opacity=".22"')+'><path d="M12 2l3 6.6 7 .8-5.2 4.8 1.4 7L12 17.8 5.8 21.2l1.4-7L2 9.4l7-.8z"/></svg>';
    return h+'</div>';
  }
  function esc(s){return (s||'').replace(/[<>&]/g,function(c){return {'<':'&lt;','>':'&gt;','&':'&amp;'}[c];});}
  function ru(d){
    var m=['янв','фев','мар','апр','мая','июн','июл','авг','сен','окт','ноя','дек'];
    var p=(d||'').split('-'); if(p.length<3)return '';
    return parseInt(p[2],10)+' '+m[parseInt(p[1],10)-1]+' '+p[0];
  }

  function summary(){
    var b=DATA.brand||{}, dist=DATA.dist||null;
    var h='<div><div class="big">'+(b.avg||0).toFixed(1).replace('.',',')+'</div>'+
          stars(Math.round(b.avg||0))+
          '<div class="cnt">'+(b.count||0).toLocaleString('ru-RU')+' отзывов</div></div>';
    if(dist){
      h+='<div class="bars">';
      for(var s=5;s>=1;s--){
        var n=dist[s]||0, pc=b.count?Math.round(n/b.count*100):0;
        h+='<div class="bar"><span>'+s+'</span><i><b style="width:'+pc+'%"></b></i><span>'+pc+'%</span></div>';
      }
      h+='</div>';
    }
    h+='<div class="src">Отзывы с Wildberries по товарам, которые продаются на сайте.'+
       (DATA.updated?'<br>Обновлено '+DATA.updated:'')+'</div>';
    $('hrvTop').innerHTML=h;
  }

  function products(){
    var list=Object.keys(DATA.reviews||{}).map(function(nm){
      var r=DATA.reviews[nm][0]||{};
      return {nm:nm,name:r.product||('арт. '+nm),n:DATA.reviews[nm].length};
    }).sort(function(a,b){return b.n-a.n;});
    var h='<button class="on" data-nm="">Все отзывы</button>';
    list.forEach(function(p){
      h+='<button data-nm="'+p.nm+'">'+esc(p.name.slice(0,34))+' · '+p.n+'</button>';
    });
    $('hrvFilt').innerHTML=h;
    $('hrvFilt').addEventListener('click',function(e){
      var b=e.target.closest('button'); if(!b)return;
      Array.prototype.forEach.call($('hrvFilt').children,function(x){x.classList.toggle('on',x===b);});
      cur=b.dataset.nm; apply();
    });
  }

  function apply(){
    var all=[];
    Object.keys(DATA.reviews||{}).forEach(function(nm){
      if(cur&&nm!==cur)return;
      all=all.concat(DATA.reviews[nm]);
    });
    // сначала с фото, дальше по дате — фото убеждают сильнее текста
    all.sort(function(a,b){
      var pa=(a.photos||[]).length>0, pb=(b.photos||[]).length>0;
      if(pa!==pb)return pb-pa;
      return (b.date||'').localeCompare(a.date||'');
    });
    filtered=all; shown=0; $('hrvGrid').innerHTML=''; render();
  }

  function render(){
    var part=filtered.slice(shown,shown+PAGE), h='';
    part.forEach(function(r){
      h+='<article class="rv"><div class="hd"><div class="who">'+esc(r.user)+'</div>'+
         '<div class="dt">'+ru(r.date)+'</div></div>'+
         stars(r.val,'st')+
         (r.text?'<p>'+esc(r.text)+'</p>':'')+
         (r.pros?'<p style="margin-top:7px"><b>Плюсы:</b> '+esc(r.pros)+'</p>':'')+
         (r.size?'<span class="sz">размер '+esc(r.size)+'</span>':'');
      if((r.photos||[]).length){
        h+='<div class="ph">';
        r.photos.forEach(function(p){h+='<img loading="lazy" src="'+p+'" alt="Фото из отзыва — '+esc(r.product)+'">';});
        h+='</div>';
      }
      h+='<div class="prod"><div><div class="nm">'+esc(r.product)+'</div>'+
         '<div class="art">арт. '+esc(r.art||r.nm)+'</div></div>'+
         (r.url?'<a href="'+r.url+'">Смотреть →</a>':'')+
         '</div></article>';
    });
    $('hrvGrid').insertAdjacentHTML('beforeend',h);
    shown+=part.length;
    $('hrvMore').style.display=shown<filtered.length?'block':'none';
    if(!filtered.length)$('hrvGrid').innerHTML='<div class="empty">Пока нет отзывов по этому товару.</div>';
  }

  $('hrvMore').addEventListener('click',render);

  // просмотр фото
  var lb=$('hrvLb');
  document.addEventListener('click',function(e){
    var img=e.target.closest('.hrv .ph img');
    if(img){lb.querySelector('img').src=img.src;lb.classList.add('on');return;}
    if(e.target.closest('.hrv .lb'))lb.classList.remove('on');
  });
  document.addEventListener('keydown',function(e){if(e.key==='Escape')lb.classList.remove('on');});

  fetch(SRC+'?t='+Date.now()).then(function(r){return r.json();}).then(function(d){
    DATA=d; summary(); products(); apply();
    // разметка для поисковиков: общая оценка магазина по нашим товарам
    try{
      var b=d.brand||{};
      if(b.count){
        var s=document.createElement('script');s.type='application/ld+json';
        s.textContent=JSON.stringify({"@context":"https://schema.org","@type":"Product",
          "name":"Женские джинсы больших размеров HAILEY","brand":{"@type":"Brand","name":"HAILEY"},
          "aggregateRating":{"@type":"AggregateRating","ratingValue":b.avg,"reviewCount":b.count,
            "bestRating":5,"worstRating":1}});
        document.head.appendChild(s);
      }
    }catch(e){}
  }).catch(function(){
    $('hrvGrid').innerHTML='<div class="empty">Отзывы временно недоступны. Загляните чуть позже.</div>';
  });
})();
</script>
