<script>
/* Сквозная шапка стоит НАД баннером, поэтому 100vh даёт экран + шапку — низ уезжает.
   Меряем шапку и вычитаем её высоту; при повороте экрана пересчитываем. */
(function(){try{
  var hero=document.querySelector('.hxC');if(!hero)return;
  function sync(){
    var h=document.querySelector('.hh');
    var v=h?Math.round(h.getBoundingClientRect().height):0;
    hero.style.setProperty('--hh', v+'px');
  }
  sync(); setTimeout(sync,300); setTimeout(sync,1200);
  window.addEventListener('resize',sync);
  window.addEventListener('orientationchange',sync);
}catch(e){}})();
</script>
