    $('tPromoCode').textContent=promo;$('tPromo').textContent='−'+rub(cut);
    $('tTotal').textContent=rub(g-cut);
    var t='Заказать · '+rub(g-cut);
    $('doOrder').textContent=t;$('doOrderM').textContent=t;
  }
  /* Промокод. Скидку тут только ПОКАЗЫВАЕМ — к оплате её пересчитывает сервер (/pay/create),
     иначе код можно было бы придумать в консоли браузера. Список кодов держать одинаковым
     здесь и в PROMOS в pay_endpoint.py. */
  var PROMOS={'HAILEY10':10};
  var promo='';
  // считаем ровно как сервер — построчно, иначе показанная скидка разойдётся с суммой к оплате
  function promoCut(g){
    var pc=PROMOS[promo]||0;
    if(!pc)return 0;
    var after=cart.reduce(function(s,i){return s+Math.max(1,Math.round(i.price*(100-pc)/100))*i.qty;},0);
    return g-after;
  }
  try{promo=(localStorage.getItem('hailey_promo_v1')||'').toUpperCase();}catch(e){}
  if(PROMOS[promo]){$('f_promo').value=promo;$('promoMsg').textContent='Промокод применён: −'+PROMOS[promo]+'%';$('promoMsg').className='pmsg ok';}
  else promo='';
  function applyPromo(){
    var v=($('f_promo').value||'').trim().toUpperCase();
    var m=$('promoMsg');
    if(!v){promo='';m.className='pmsg';render();return;}
    if(PROMOS[v]){promo=v;try{localStorage.setItem('hailey_promo_v1',v);}catch(e){}
      m.textContent='Промокод применён: −'+PROMOS[v]+'%';m.className='pmsg ok';
      promoChecked='';checkPromo(($('f_phone').value||'').replace(/\D/g,''));}
    else{promo='';m.textContent='Такого промокода нет';m.className='pmsg no';}
    render();
  }

    save();render();
  });

  // телефон: +7 и цифры
  var ph=$('f_phone');
  ph.addEventListener('input',function(){
    var d=this.value.replace(/\D/g,'');
    if(d[0]==='8')d='7'+d.slice(1);
    if(d[0]!=='7')d='7'+d;
    d=d.slice(0,11);
    var s='+7';
    if(d.length>1)s+=' '+d.slice(1,4);
    if(d.length>4)s+=' '+d.slice(4,7);
    if(d.length>7)s+='-'+d.slice(7,9);
    if(d.length>9)s+='-'+d.slice(9,11);
    this.value=s;
    checkPromo(d);
  });

  /* Промокод действует на первый заказ ЧЕЛОВЕКА, а лежит он в памяти УСТРОЙСТВА — поэтому
     у постоянного покупателя подставляется сам. Как только телефон введён полностью,
     спрашиваем сервер (он же считает сумму к оплате) и снимаем скидку, если она не положена:
     иначе в корзине одна цена, а на странице банка другая. */
  var promoChecked='';
  function checkPromo(digits){
    if(!promo||digits.length<11||promoChecked===digits)return;
    promoChecked=digits;
    fetch(PAY+'/promo',{method:'POST',headers:{'Content-Type':'application/json'},
      body:JSON.stringify({phone:digits})})
      .then(function(r){return r.json();})
      .then(function(d){
        if(!d||d.eligible!==false)return;
        promo='';try{localStorage.removeItem('hailey_promo_v1');}catch(e){}
        $('f_promo').value='';
        $('promoMsg').textContent='Промокод действует только на первый заказ';
        $('promoMsg').className='pmsg no';
        render();
      }).catch(function(){});
  }

  function orderText(){
    var L=cart.map(function(i){return '• '+i.name+(i.size?' (размер '+i.size+')':'')+' × '+i.qty+' = '+rub(i.price*i.qty);});
    L.push('———');
    L.push('ИТОГО: '+rub(goods()-promoCut(goods())));
    L.push('Доставка: '+dlv+(($('f_city').value||'').trim()?', '+$('f_city').value.trim():'')+(($('f_addr').value||'').trim()?', '+$('f_addr').value.trim():''));
    L.push('Оплата: '+pay);
    if(promo&&PROMOS[promo])L.push('Промокод: '+promo+' (−'+PROMOS[promo]+'%)');
    var c=($('f_cmt').value||'').trim();if(c)L.push('Комментарий: '+c);
    return 'Заказ HAILEY (сайт)\n'+L.join('\n');
  }
  function nativeForm(){var fs=document.querySelectorAll('.js-form-proccess');for(var i=0;i<fs.length;i++){if(fs[i].querySelector('[name="Phone"]'))return fs[i];}return fs[0]||null;}

  function submit(){
    var name=($('f_name').value||'').trim(), phone=($('f_phone').value||'').trim();
    $('err1').style.display='none';$('f_name').classList.remove('err');$('f_phone').classList.remove('err');
    if(name.length<2)return fail('Впишите, пожалуйста, имя','f_name');
    if(phone.replace(/\D/g,'').length<11)return fail('Впишите телефон полностью','f_phone');
    var f=nativeForm();
    if(f){
      function set(n,v){var el=f.querySelector('[name="'+n+'"]');if(el){el.value=v;['input','change','blur'].forEach(function(ev){el.dispatchEvent(new Event(ev,{bubbles:true}));});}}
      set(f.querySelector('[name="zakaz"]')?'zakaz':'form-spec-comments',orderText());
      set('Name',name);set('Phone',phone);
      var em=f.querySelector('[name="Email"]');if(em&&!em.value)set('Email','order+'+phone.replace(/\D/g,'')+'@hailey.site');
      var cm=($('f_cmt').value||'').trim();if(cm)set('Comments',cm);
      var b=f.querySelector('.t-submit, .t-form__submit button, button[type=submit]');
      if(b)b.click();else f.dispatchEvent(new Event('submit',{bubbles:true,cancelable:true}));
    }
    var num='HL-'+String(new Date().getTime()).slice(-6);
    var snapshot=cart.slice();
    cart=[];save();
    // Заказ уже ушёл в «Заявки». Дальше пробуем поднять оплату; не вышло — обычный экран «Заказ принят».
    [$('doOrder'),$('doOrderM')].forEach(function(b){if(b){b.disabled=true;b.textContent='Переходим к оплате…';}});
    var done=false;
    function finish(){if(done)return;done=true;showOk(num);}
    var t=setTimeout(finish,12000);
    try{
      fetch(PAY+'/create',{method:'POST',headers:{'Content-Type':'application/json'},
        body:JSON.stringify({order_id:num,name:name,phone:phone,promo:promo,
          method:(pay==='СБП'?'sbp':'card'),
          items:snapshot.map(function(i){return{id:i.id,qty:i.qty,size:i.size||''};})})})
      .then(function(r){return r.json();})
      .then(function(d){
        if(done)return;
        // сервер срезал скидку как повторную — стираем код, чтобы корзина больше не рисовала
        // цену, которой не будет: настоящая сумма всё равно видна на странице банка
        if(d&&d.promo_used===false&&promo){try{localStorage.removeItem('hailey_promo_v1');}catch(e){}}
        if(d&&d.redirect){done=true;clearTimeout(t);location.href=d.redirect;}
        else{clearTimeout(t);finish();}
      })
      .catch(function(){clearTimeout(t);finish();});
    }catch(e){clearTimeout(t);finish();}
  }
