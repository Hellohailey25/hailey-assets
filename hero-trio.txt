<script>
/* Сквозная шапка стоит НАД баннером: 100vh даёт экран + шапку, низ уезжает.
   Меряем шапку и вычитаем её высоту; при повороте экрана пересчитываем. */
(function(){try{
  var hero=document.querySelector('.hxD');if(!hero)return;
  function sync(){
    var h=document.querySelector('.hh');
    hero.style.setProperty('--hh',(h?Math.round(h.getBoundingClientRect().height):0)+'px');
  }
  sync(); setTimeout(sync,300); setTimeout(sync,1200);
  window.addEventListener('resize',sync);
  window.addEventListener('orientationchange',sync);
}catch(e){}})();
</script>
