<script>
(function(){try{
  var KEY='hailey_cart_v1';
  function load(){try{var a=JSON.parse(localStorage.getItem(KEY));return Array.isArray(a)?a:[];}catch(e){return[];}}
  function save(a){try{localStorage.setItem(KEY,JSON.stringify(a));}catch(e){}}
  function rub(n){return n.toLocaleString('ru-RU')+' руб';}

  function paint(){
    var c=load(), el=document.getElementById('toCart');
    if(!c.length){el.textContent='Корзина пуста';return;}
    var sum=c.reduce(function(s,i){return s+i.price*i.qty;},0);
    var n=c.reduce(function(s,i){return s+i.qty;},0);
    el.innerHTML='В корзине <b>'+n+'</b> шт на <b>'+rub(sum)+'</b>: '+
      c.map(function(i){return i.name+' · '+i.size+(i.qty>1?' ×'+i.qty:'');}).join(', ');
  }

  [].forEach.call(document.querySelectorAll('.hto .prod'),function(box){
    var sizes=box.querySelectorAll('.sz'), btn=box.querySelector('.add'), chosen=null;
    [].forEach.call(sizes,function(b){
      b.addEventListener('click',function(){
        [].forEach.call(sizes,function(x){x.classList.remove('on');});
        b.classList.add('on'); chosen=b.textContent.trim();
        btn.classList.add('ready'); btn.textContent='В корзину · размер '+chosen;
      });
    });
    btn.addEventListener('click',function(){
      if(!chosen)return;
      var c=load(), id=box.dataset.id, name=box.dataset.name;
      // структура позиции ровно как в карточке товара, иначе корзина и оформление её не поймут
      var ex=c.filter(function(i){return i.id===id&&i.size===chosen;})[0];
      if(ex) ex.qty++;
      else c.push({id:id,name:name,price:+box.dataset.price,img:'',size:chosen,qty:1});
      save(c); paint();
      btn.textContent='Добавлено · ещё раз?';
      setTimeout(function(){btn.textContent='В корзину · размер '+chosen;},1600);
    });
  });

  document.getElementById('toClear').addEventListener('click',function(){save([]);paint();});
  paint();
}catch(e){}})();
</script>
