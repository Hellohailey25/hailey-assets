<script>
(function(){
  var FAV_KEY='hailey_fav_v1';
  var grid=document.getElementById('zgrid'),empty=document.getElementById('zempty'),
      cnt=document.getElementById('zcnt'),lead=document.getElementById('zlead'),toast=document.getElementById('ztoast');
  var rub=function(n){return (n||0).toLocaleString('ru-RU')+' руб';};
  function load(){try{var a=JSON.parse(localStorage.getItem(FAV_KEY));return Array.isArray(a)?a:[];}catch(e){return[];}}
  function save(a){try{localStorage.setItem(FAV_KEY,JSON.stringify(a));}catch(e){}}
  function plural(n){var a=Math.abs(n)%100,b=n%10,w;if(a>10&&a<20)w='товаров';else if(b===1)w='товар';else if(b>=2&&b<=4)w='товара';else w='товаров';return n+' '+w;}
  var tt;function showToast(m){toast.textContent=m;toast.classList.add('on');clearTimeout(tt);tt=setTimeout(function(){toast.classList.remove('on');},2200);}

  function render(){
    var list=load();
    cnt.textContent=list.length?plural(list.length):'';
    lead.textContent=list.length?'Товары сохранены в этом браузере — можно вернуться к ним позже.':'';
    empty.style.display=list.length?'none':'block';
    grid.innerHTML='';
    list.forEach(function(p){
      var el=document.createElement('div');
      el.className='card';
      var link=p.page||'/catalog';
      el.innerHTML=
        '<div class="ph">'
        +'<a href="'+link+'"><img src="'+(p.img||'')+'" alt="'+(p.name||'')+'"></a>'
        +'<button class="fav" data-rm="'+p.id+'" aria-label="Убрать из избранного"><svg viewBox="0 0 24 24"><path d="M12 20.7l-1.4-1.3C5.4 14.6 2 11.5 2 7.8 2 5 4.2 3 6.9 3c1.6 0 3.1.7 4.1 1.9C12 3.7 13.5 3 15.1 3 17.8 3 20 5 20 7.8c0 3.7-3.4 6.8-8.6 11.6L12 20.7z"/></svg></button>'
        +'</div>'
        +'<div class="info"><a class="pname" href="'+link+'">'+(p.name||'')+'</a>'
        +'<div class="prices"><span class="price">'+rub(p.price)+'</span>'
        +((p.old&&p.old>p.price)?'<span class="old">'+rub(p.old)+'</span>':'')+'</div>'
        +'<a class="go" href="'+link+'">Смотреть товар</a></div>';
      grid.appendChild(el);
    });
  }
  grid.addEventListener('click',function(e){
    var b=e.target.closest('[data-rm]');if(!b)return;
    e.preventDefault();
    var id=b.dataset.rm,a=load(),i=-1;
    a.forEach(function(x,n){if(x.id===id)i=n;});
    if(i>-1){a.splice(i,1);save(a);render();showToast('Убрано из избранного');}
  });
  window.addEventListener('storage',function(e){if(e.key===FAV_KEY)render();});
  render();
})();
</script>
